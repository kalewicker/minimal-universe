cmake_minimum_required(VERSION 3.20)
project(minimal_universe_g2_calibration LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(MU_CAL_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
add_executable(mu-g2-calibration main.cpp ../g2_assay/kernel.cpp)
target_include_directories(mu-g2-calibration PRIVATE ../g2_assay)
target_compile_options(mu-g2-calibration PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(MU_CAL_SANITIZE)
 target_compile_options(mu-g2-calibration PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(mu-g2-calibration PRIVATE -fsanitize=address,undefined)
endif()
