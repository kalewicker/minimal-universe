cmake_minimum_required(VERSION 3.20)
project(minimal_universe_g2_assay LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(MU_ASSAY_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
add_executable(mu-g2-assay kernel.cpp main.cpp)
target_compile_options(mu-g2-assay PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(MU_ASSAY_SANITIZE)
  target_compile_options(mu-g2-assay PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(mu-g2-assay PRIVATE -fsanitize=address,undefined)
endif()
