cmake_minimum_required(VERSION 3.20)
project(minimal_universe_g2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(MU_G2_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
add_library(g2_kernel kernel.cpp)
target_include_directories(g2_kernel PUBLIC .)
target_compile_options(g2_kernel PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(MU_G2_SANITIZE)
  target_compile_options(g2_kernel PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(g2_kernel PUBLIC -fsanitize=address,undefined)
endif()
add_executable(mu-g2 main.cpp)
target_link_libraries(mu-g2 PRIVATE g2_kernel)
enable_testing()
add_executable(g2_tests tests.cpp)
target_link_libraries(g2_tests PRIVATE g2_kernel)
add_test(NAME g2_kernel COMMAND g2_tests)
