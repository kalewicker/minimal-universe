cmake_minimum_required(VERSION 3.20)
project(minimal_universe VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(MSRU_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
add_library(kernels src/graph.cpp src/ski.cpp)
target_include_directories(kernels PUBLIC include)
target_compile_options(kernels PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(MSRU_SANITIZE)
  target_compile_options(kernels PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(kernels PUBLIC -fsanitize=address,undefined)
endif()
add_executable(msru src/main.cpp)
target_link_libraries(msru PRIVATE kernels)
enable_testing()
add_executable(kernel_tests tests/kernel_tests.cpp)
target_link_libraries(kernel_tests PRIVATE kernels)
add_test(NAME kernels COMMAND kernel_tests)
